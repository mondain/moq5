cmake_minimum_required(VERSION 3.20)
project(packager_consumer LANGUAGES C CXX)
find_package(libmoq CONFIG REQUIRED)
foreach(lang c cpp)
    add_executable(consumer_${lang} main.${lang})
    target_link_libraries(consumer_${lang} PRIVATE moq::cmaf-packager)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
        target_compile_options(consumer_${lang} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
target_compile_features(consumer_c PRIVATE c_std_11)
target_compile_features(consumer_cpp PRIVATE cxx_std_17)
