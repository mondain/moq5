cmake_minimum_required(VERSION 3.20)
get_filename_component(moq_source "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)

# Child mode receives a producer preload; child-local tests root registration.
if(MODE STREQUAL "child-local")
    project(MoqAppleAbiProbeFixture LANGUAGES NONE)
    if(DEFINED CMAKE_USER_MAKE_RULES_OVERRIDE_C OR DEFINED CMAKE_USER_MAKE_RULES_OVERRIDE_CXX)
        message(FATAL_ERROR "Child-local fixture requires undefined parent language hooks")
    endif()
    set(CMAKE_USER_MAKE_RULES_OVERRIDE "")
    add_subdirectory(child-local)
    add_subdirectory(sibling)
    if(NOT DEFINED CMAKE_USER_MAKE_RULES_OVERRIDE
       OR NOT CMAKE_USER_MAKE_RULES_OVERRIDE STREQUAL ""
       OR DEFINED CMAKE_USER_MAKE_RULES_OVERRIDE_C
       OR DEFINED CMAKE_USER_MAKE_RULES_OVERRIDE_CXX)
        message(FATAL_ERROR "Library changed parent hooks")
    endif()
    return()
endif()

if(MODE STREQUAL "named" OR MODE STREQUAL "relative")
    list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}")
    if(MODE STREQUAL "named")
        set(CMAKE_USER_MAKE_RULES_OVERRIDE_CXX DelayedRules)
    else()
        set(CMAKE_USER_MAKE_RULES_OVERRIDE_CXX "./DelayedRules.cmake")
    endif()
endif()
if(MODE STREQUAL "caller" OR MODE STREQUAL "embed")
    set(caller_hook "${CMAKE_CURRENT_LIST_DIR}/CallerRules.cmake")
    set(CMAKE_USER_MAKE_RULES_OVERRIDE "${caller_hook}")
    set(CMAKE_USER_MAKE_RULES_OVERRIDE_C "${caller_hook}")
    set(CMAKE_USER_MAKE_RULES_OVERRIDE_CXX "${caller_hook}")
endif()
if(NOT MODE STREQUAL "child" AND NOT CMAKE_TOOLCHAIN_FILE
   AND "$ENV{CMAKE_TOOLCHAIN_FILE}" STREQUAL "")
    include("${moq_source}/cmake/MoqAppleAbiProbe.cmake")
endif()
project(MoqAppleAbiProbeFixture LANGUAGES C)

if(MODE STREQUAL "named" OR MODE STREQUAL "relative")
    include("${moq_source}/tests/cmake/ConsumerBuildContext.cmake")
    moq_write_consumer_abi_context("${CMAKE_CURRENT_BINARY_DIR}/delayed-context.cmake")
    if(delayed_rules_seen)
        message(FATAL_ERROR "Serializer executed delayed CXX hook")
    endif()
endif()

if(MODE STREQUAL "embed" OR MODE STREQUAL "toolchain")
    set(MOQ_BUILD_TESTS OFF)
    set(MOQ_BUILD_SIM OFF)
    set(MOQ_BUILD_EXAMPLES OFF)
    add_subdirectory("${moq_source}" libmoq)
endif()
enable_language(CXX)
if((MODE STREQUAL "named" OR MODE STREQUAL "relative" OR EXPECT_DELAYED)
   AND NOT delayed_rules_seen)
    message(FATAL_ERROR "Delayed CXX hook did not execute")
endif()

if(MODE STREQUAL "toolchain")
    if(NOT toolchain_rules_selected)
        message(FATAL_ERROR "Toolchain did not select its own hooks")
    endif()
    set(caller_hook "${CMAKE_CURRENT_LIST_DIR}/CallerRules.cmake")
endif()

if(DEFINED caller_hook)
    if(NOT caller_rules_seen)
        message(FATAL_ERROR "Caller hook did not execute")
    endif()
    foreach(var CMAKE_USER_MAKE_RULES_OVERRIDE CMAKE_USER_MAKE_RULES_OVERRIDE_C
            CMAKE_USER_MAKE_RULES_OVERRIDE_CXX)
        if(NOT "${${var}}" STREQUAL "${caller_hook}")
            message(FATAL_ERROR "Caller hook replaced: ${var}")
        endif()
    endforeach()
endif()

set(evidence "")
foreach(lang C CXX)
    if(NOT CMAKE_${lang}_ABI_COMPILED)
        message(FATAL_ERROR "${lang} ABI discovery failed")
    endif()
    if(APPLE AND CMAKE_${lang}_COMPILER_ID MATCHES "^(AppleClang|Clang)$"
       AND CMAKE_${lang}_VERBOSE_FLAG STREQUAL "-v -Wl,-v")
        if(NOT CMAKE_${lang}_VERBOSE_COMPILE_FLAG STREQUAL "-v")
            message(FATAL_ERROR "${lang} compile verbosity is not corrected")
        endif()
    endif()
    foreach(field COMPILER COMPILER_ID COMPILER_VERSION SIZEOF_DATA_PTR
            BYTE_ORDER COMPILER_ABI IMPLICIT_INCLUDE_DIRECTORIES
            IMPLICIT_LINK_LIBRARIES IMPLICIT_LINK_DIRECTORIES
            IMPLICIT_LINK_FRAMEWORK_DIRECTORIES LIBRARY_ARCHITECTURE
            VERBOSE_FLAG VERBOSE_COMPILE_FLAG)
        string(APPEND evidence "CMAKE_${lang}_${field}=[${CMAKE_${lang}_${field}}]\n")
    endforeach()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/abi-evidence.txt" "${evidence}")
include("${moq_source}/tests/cmake/ConsumerBuildContext.cmake")
moq_write_consumer_context("${CMAKE_CURRENT_BINARY_DIR}/full-context.cmake")
moq_write_consumer_abi_context("${CMAKE_CURRENT_BINARY_DIR}/abi-context.cmake")
